3. Operators & Coercion

// == coerces, === does not
console.log(1 == "1", 1 === "1", null == undefined, null === undefined);

// + prefers strings, other math operators prefer numbers
console.log("5" + 3, "5" - 3, "5" * "2", +"42", +"abc");

// Logical operators return operands
console.log(0 || "default", "" ?? "default", null ?? "default");
console.log(1 && 2, 0 && 2);

// Optional chaining + nullish coalescing
const cfg = { server: { port: 0 } };
console.log(cfg.server?.port ?? 8080);     // 0 (?? only checks null/undefined)
console.log(cfg.db?.host ?? "localhost");
console.log(cfg.run?.());                  // undefined, no error

// Logical assignment
let count = null;
count ??= 10;  count ||= 20;  count &&= 30;
console.log(count);

// Exponent, increment, bitwise
console.log(2 ** 10, 7 % 3, 5 & 3, 5 | 3, 5 ^ 3, 1 << 4);
Prefer === / !==. Use ?? instead of || when 0, "" or false are valid values.

10. Closures

A closure is a function bundled with references to the variables in its lexical scope, even after the outer function has returned.

function counter() {
  let count = 0;                        // private state
  return {
    inc: () => ++count,
    dec: () => --count,
    get value() { return count; },
  };
}
const c = counter();
c.inc(); c.inc(); c.dec();
console.log(c.value);

// classic loop pitfall
const fnsVar = [];
for (var i = 0; i < 3; i++) fnsVar.push(() => i);
console.log(fnsVar.map((f) => f()));       // [3,3,3]

const fnsLet = [];
for (let j = 0; j < 3; j++) fnsLet.push(() => j);
console.log(fnsLet.map((f) => f()));       // [0,1,2]

// memoize
function memoize(fn) {
  const cache = new Map();
  return (n) => cache.has(n) ? cache.get(n) : (cache.set(n, fn(n)), cache.get(n));
}
const slowSquare = (n) => { console.log("computing", n); return n * n; };
const fast = memoize(slowSquare);
console.log(fast(4), fast(4));

// debounce
function debounce(fn, ms) {
  let t;
  return (...args) => { clearTimeout(t); t = setTimeout(() => fn(...args), ms); };
}
const log = debounce((m) => console.log("debounced:", m), 50);
log("a"); log("b"); log("c");

// classes are syntax over prototypes
class Shape {
  static count = 0;
  #id;                                    // private field
  constructor(name) { this.name = name; this.#id = ++Shape.count; }
  get id() { return this.#id; }
  area() { return 0; }
  toString() { return `${this.name}#${this.#id} area=${this.area().toFixed(2)}`; }
  static create(name) { return new this(name); }
}

class Circle extends Shape {
  constructor(r) { super("Circle"); this.r = r; }
  area() { return Math.PI * this.r ** 2; }
}

class Square extends Shape {
  constructor(s) { super("Square"); this.s = s; }
  area() { return this.s ** 2; }
}

const shapes = [new Circle(2), new Square(3)];
shapes.forEach((s) => console.log(String(s)));

15. Async: Callbacks, Promises & async/await

const wait = (ms, value, fail = false) =>
  new Promise((resolve, reject) => setTimeout(() => (fail ? reject(new Error(value)) : resolve(value)), ms));

// promise chain
await wait(30, "chain start")
  .then((v) => { console.log(v); return wait(30, "second"); })
  .then((v) => { console.log(v); throw new Error("boom"); })
  .catch((e) => console.log("caught:", e.message))
  .finally(() => console.log("finally"));

// async / await + try/catch
async function load() {
  try {
    const a = await wait(20, "A");
    const b = await wait(20, "B", true);
    return a + b;
  } catch (e) {
    return "failed: " + e.message;
  }
}
console.log(await load());

// sequential vs parallel
console.time("sequential");
await wait(100); await wait(100);
console.timeEnd("sequential");

console.time("parallel");
await Promise.all([wait(100), wait(100)]);
console.timeEnd("parallel");

// combinators
console.log(await Promise.all([wait(10, 1), wait(20, 2)]));
console.log(await Promise.allSettled([wait(10, "ok"), wait(10, "bad", true)]));
console.log(await Promise.race([wait(50, "slow"), wait(10, "fast")]));
console.log(await Promise.any([wait(10, "fail", true), wait(30, "winner")]));

// timeout helper
const withTimeout = (p, ms) => Promise.race([p, new Promise((_, rej) => setTimeout(() => rej(new Error("timeout")), ms))]);
try { await withTimeout(wait(200), 50); } catch (e) { console.log(e.message); }
This runner wraps code in an async function, so top-level await works in the examples.

17. Error Handling

class ValidationError extends Error {
  constructor(message, field) {
    super(message);
    this.name = "ValidationError";
    this.field = field;
  }
}

function validate(user) {
  if (!user.email) throw new ValidationError("Email required", "email");
  return true;
}

try {
  validate({});
} catch (err) {
  if (err instanceof ValidationError) console.log(`${err.name} on ${err.field}: ${err.message}`);
  else throw err;
} finally {
  console.log("cleanup runs always");
}

// error cause
try {
  try { JSON.parse("{bad"); }
  catch (e) { throw new Error("Config invalid", { cause: e }); }
} catch (e) {
  console.log(e.message, "->", e.cause.name);
}

// built-in error types
for (const fn of [() => null.x, () => undefinedVar, () => new Array(-1), () => decodeURI("%")]) {
  try { fn(); } catch (e) { console.log(e.constructor.name); }
}

18. Modules (ES Modules)

Use <script type="module"> in browsers or "type": "module" in Node's package.json. Modules are strict mode, have their own scope and load once.

// dynamic import (lazy loading)
const { add } = await import("./math.js");

    23. Common Pitfalls

    // 1. sort() default is lexicographic
console.log([2, 10, 1].sort());

// 2. parseInt with map
console.log(["1", "2", "3"].map(parseInt));          // [1, NaN, NaN]
console.log(["1", "2", "3"].map(Number));

// 3. typeof null
console.log(typeof null);

// 4. array equality is by reference
console.log([1] == [1], JSON.stringify([1]) === JSON.stringify([1]));

// 5. forEach does not await
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
[1, 2].forEach(async (n) => { await sleep(10); });
console.log("forEach finished immediately; use for...of or Promise.all");

// 6. mutating shared defaults
function bad(list = []) { list.push(1); return list; }
console.log(bad(), bad());                            // fresh array each call (OK in JS)

// 7. floating point money
console.log(0.1 * 3, (0.1 * 3).toFixed(2), Math.round(0.1 * 3 * 100) / 100);

// 8. NaN checks
console.log(isNaN("abc"), Number.isNaN("abc"));

// 9. shallow copy gotcha
const orig = { nested: { v: 1 } };
const copy = { ...orig };
copy.nested.v = 99;
console.log(orig.nested.v);

// 10. object keys are strings
const o = {}; o[1] = "a"; o["1"] = "b";
console.log(o);

    4. Implement Promise.all

    const myAll = (promises) => new Promise((resolve, reject) => {
  const results = []; let done = 0;
  if (!promises.length) return resolve(results);
  promises.forEach((p, i) => Promise.resolve(p).then((v) => {
    results[i] = v;
    if (++done === promises.length) resolve(results);
  }, reject));
});

    6. Deep equality

    const deepEqual = (a, b) => {
  if (a === b) return true;
  if (typeof a !== "object" || typeof b !== "object" || !a || !b) return false;
  const ka = Object.keys(a), kb = Object.keys(b);
  return ka.length === kb.length && ka.every((k) => deepEqual(a[k], b[k]));
};
console.log(deepEqual({ a: [1, { b: 2 }] }, { a: [1, { b: 2 }] }), deepEqual({ a: 1 }, { a: 2 }));
