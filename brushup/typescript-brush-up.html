TypeScript Brush-Up

TypeScript is JavaScript plus a static type system. Types are erased at compile time, so there is zero runtime cost. Click Run to transpile and execute an example in your browser (the compiler is downloaded from a CDN on first click). Blocks with a red left border show code that fails to compile; they are not runnable.

The in-browser runner only transpiles (strips types). Use tsc or your editor to see real type errors.

1. Setup & Compiler

# install

# compile

npx tsc file.ts                # single file

// optional, default, rest

8. Narrowing & Type Guards

// typeof, instanceof, in, equality, truthiness
function describe(x: string | number | Date | string[] | null): string {
  if (x === null) return "null";
  if (typeof x === "string") return `string:${x.length}`;
  if (typeof x === "number") return `number:${x.toFixed(1)}`;
  if (x instanceof Date) return `date:${x.getUTCFullYear()}`;
  return `array:${x.length}`;
}
console.log(describe("hi"), describe(1), describe(new Date(0)), describe(["a"]), describe(null));

// custom type guard
interface Cat { meow(): void }
interface Dog { bark(): void }
function isCat(pet: Cat | Dog): pet is Cat {
  return (pet as Cat).meow !== undefined;
}
const pet: Cat | Dog = { meow() { console.log("meow"); } };
if (isCat(pet)) pet.meow();

// discriminated union + exhaustive check
type Shape =
  | { kind: "circle"; radius: number }
  | { kind: "rect"; w: number; h: number }
  | { kind: "square"; size: number };

function area(s: Shape): number {
  switch (s.kind) {
    case "circle": return Math.PI * s.radius ** 2;
    case "rect": return s.w * s.h;
    case "square": return s.size ** 2;
    default: {
      const _exhaustive: never = s;      // compile error if a case is missing
      return _exhaustive;
    }
  }
}
console.log(area({ kind: "circle", radius: 1 }).toFixed(2), area({ kind: "rect", w: 2, h: 3 }));

// assertion function
function assertString(v: unknown): asserts v is string {
  if (typeof v !== "string") throw new TypeError("not a string");
}
const val: unknown = "ok";
assertString(val);
console.log(val.toUpperCase());

9. Generics

function identity<T>(value: T): T { return value; }
console.log(identity<string>("a"), identity(42));       // T inferred

// constraints
function longest<T extends { length: number }>(a: T, b: T): T {
  return a.length >= b.length ? a : b;
}
console.log(longest("abc", "de"), longest([1, 2], [1]));

// keyof constraint
function pluck<T, K extends keyof T>(items: T[], key: K): T[K][] {
  return items.map((i) => i[key]);
}
console.log(pluck([{ id: 1, n: "a" }, { id: 2, n: "b" }], "n"));

// generic interfaces & classes
interface Box<T> { value: T }
class Stack<T> {
  private items: T[] = [];
  push(item: T): void { this.items.push(item); }
  pop(): T | undefined { return this.items.pop(); }
  peek(): T | undefined { return this.items[this.items.length - 1]; }
  get size(): number { return this.items.length; }
}
const s = new Stack<number>();
s.push(1); s.push(2);
console.log(s.pop(), s.peek(), s.size);

// default type parameter
type ApiResponse<T = unknown> = { ok: boolean; data: T };
const res: ApiResponse<string[]> = { ok: true, data: ["a"] };
console.log(res);

// generic factory
function create<T>(Ctor: new () => T): T { return new Ctor(); }
class Foo { hello = "foo"; }
console.log(create(Foo).hello);

// Result type
type Result<T, E = Error> = { ok: true; value: T } | { ok: false; error: E };
function safeParse(json: string): Result<unknown> {
  try { return { ok: true, value: JSON.parse(json) }; }
  catch (e) { return { ok: false, error: e as Error }; }
}

10. Classes

interface Printable { print(): string }

abstract class Shape implements Printable {
  static count = 0;
  protected readonly id: number;

  // parameter properties: declare + assign in one go
  constructor(public readonly name: string, private color: string = "black") {
    this.id = ++Shape.count;
  }
  abstract area(): number;
  print(): string { return `${this.name}#${this.id} (${this.color}) area=${this.area().toFixed(2)}`; }
}

class Circle extends Shape {
  constructor(private radius: number) { super("Circle", "red"); }
  area(): number { return Math.PI * this.radius ** 2; }
}

class Rect extends Shape {
  constructor(private w: number, private h: number) { super("Rect"); }
  area(): number { return this.w * this.h; }
  override print(): string { return "[Rect] " + super.print(); }
}

const shapes: Shape[] = [new Circle(2), new Rect(2, 3)];
shapes.forEach((s) => console.log(s.print()));
console.log(Shape.count);

// JS private fields (#) are enforced at runtime too
class Account {
  #balance = 0;
  deposit(n: number): this { this.#balance += n; return this; }
  get balance(): number { return this.#balance; }
}
console.log(new Account().deposit(10).deposit(5).balance);

// accessors and index signatures in classes
class Temp {
  constructor(private _c: number) {}
  get f(): number { return this._c * 9 / 5 + 32; }
  set f(v: number) { this._c = (v - 32) * 5 / 9; }
}
const t = new Temp(100);
console.log(t.f);
new Shape("x");                   // error cannot instantiate abstract class
new Circle(1).radius;             // error 'radius' is private
new Circle(1).name = "x";         // error read-only

11. Utility Types

interface Todo { id: number; title: string; done: boolean; tags: string[] }

type A = Partial<Todo>;                          // all optional
type B = Required<Partial<Todo>>;                // all required
type C = Readonly<Todo>;                         // all readonly
type D = Pick<Todo, "id" | "title">;             // subset
type E = Omit<Todo, "tags">;                     // everything except
type F = Record<"a" | "b", number>;              // { a: number; b: number }
type G = Exclude<"a" | "b" | "c", "a">;          // "b" | "c"
type H = Extract<"a" | "b" | "c", "a" | "c">;    // "a" | "c"
type I = NonNullable<string | null | undefined>; // string

function makeTodo(): Todo { return { id: 1, title: "x", done: false, tags: [] }; }
type J = ReturnType<typeof makeTodo>;            // Todo
type K = Parameters<(a: number, b: string) => void>;   // [number, string]
type L = Awaited<Promise<Promise<number>>>;      // number

// practical use
function updateTodo(todo: Todo, changes: Partial<Todo>): Todo {
  return { ...todo, ...changes };
}

12. keyof, typeof, Mapped & Conditional Types

const config = { host: "localhost", port: 8080, debug: true };

type Config = typeof config;                       // { host: string; port: number; debug: boolean }
type ConfigKey = keyof Config;                     // "host" | "port" | "debug"
type PortType = Config["port"];                    // number  (indexed access)

// mapped types
type Nullable<T> = { [K in keyof T]: T[K] | null };
type Getters<T> = { [K in keyof T as `get${Capitalize<string & K>}`]: () => T[K] };
type MutableAll<T> = { -readonly [K in keyof T]-?: T[K] };

type CfgGetters = Getters<Config>;     // { getHost: () => string; getPort: ...; getDebug: ... }

// conditional types
type IsString<T> = T extends string ? "yes" : "no";
type T1 = IsString<"a">;       // "yes"
type T2 = IsString<1>;         // "no"

// distributes over unions
type ToArray<T> = T extends unknown ? T[] : never;
type T3 = ToArray<string | number>;        // string[] | number[]

// infer
type ElementOf<T> = T extends (infer U)[] ? U : never;
type UnwrapPromise<T> = T extends Promise<infer U> ? U : T;
type T4 = ElementOf<boolean[]>;            // boolean
type T5 = UnwrapPromise<Promise<string>>;  // string

// runtime usage of keyof
function getProp<T, K extends keyof T>(obj: T, key: K): T[K] { return obj[key]; }
console.log(getProp(config, "port"));

// recursive type
type Json = string | number | boolean | null | Json[] | { [k: string]: Json };
const j: Json = { a: [1, "b", { c: null }] };
type DeepReadonly<T> = { readonly [K in keyof T]: T[K] extends object ? DeepReadonly<T[K]> : T[K] };
const dr: DeepReadonly<{ a: { b: number } }> = { a: { b: 1 } };
console.log(j, dr);

13. Assertions, as const & satisfies

// type assertion: you override the compiler (no runtime check!)
const el = { value: "42" } as { value: string };
const n = Number(el.value);

// as const: deep readonly + literal types
const COLORS = ["red", "green", "blue"] as const;
type Color = typeof COLORS[number];              // "red" | "green" | "blue"
const ROUTES = { home: "/", about: "/about" } as const;
type Route = typeof ROUTES[keyof typeof ROUTES]; // "/" | "/about"
console.log(COLORS, ROUTES.home, n);

// satisfies: validate a value against a type without widening it
type Palette = Record<string, string | [number, number, number]>;
const palette = {
  red: [255, 0, 0],
  green: "#00ff00",
} satisfies Palette;
console.log(palette.red.map((c) => c / 255), palette.green.toUpperCase());  // precise types kept

// non-null assertion (!) - use sparingly
const maybe: string | undefined = "hi";
console.log(maybe!.length);

// definite assignment
class Svc { ready!: boolean; init() { this.ready = true; } }

// unique symbol / branded types
type UserId = number & { readonly __brand: "UserId" };
const asUserId = (n: number) => n as UserId;
function getUser(id: UserId) { return `user ${id}`; }
console.log(getUser(asUserId(5)));
getUser(5);                       // error number is not assignable to UserId (brand missing)
const x = "a" as number;          // error types do not sufficiently overlap

15. Async & Promises

interface Post { id: number; title: string }

const sleep = (ms: number): Promise<void> => new Promise((r) => setTimeout(r, ms));

async function fetchPost(id: number): Promise<Post> {
  await sleep(20);
  if (id < 0) throw new RangeError("bad id");
  return { id, title: `Post ${id}` };
}

const post = await fetchPost(1);
console.log(post.title);

// Promise.all keeps tuple types
const [p1, p2, count] = await Promise.all([fetchPost(1), fetchPost(2), Promise.resolve(10)]);
console.log(p1.title, p2.title, count + 1);

// typed error handling
try {
  await fetchPost(-1);
} catch (e: unknown) {
  if (e instanceof RangeError) console.log("RangeError:", e.message);
}

// generic retry helper
async function retry<T>(fn: () => Promise<T>, attempts = 3): Promise<T> {
  let lastErr: unknown;
  for (let i = 0; i < attempts; i++) {
    try { return await fn(); } catch (e) { lastErr = e; console.log("retrying", i + 1); }
  }
  throw lastErr;
}
let calls = 0;
console.log(await retry(async () => { if (++calls < 3) throw new Error("flaky"); return "ok after " + calls; }));

// typed fetch wrapper
async function getJson<T>(url: string): Promise<T> {
  const res = await fetch(url);
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return (await res.json()) as T;       // assertion: runtime shape is NOT verified
}
const remote = await getJson<Post>("https://jsonplaceholder.typicode.com/posts/1");
console.log(remote.title);
Generic res.json() as T is trust, not validation. For untrusted data use a runtime validator such as Zod or a custom type guard.

16. Error Handling

// With useUnknownInCatchVariables (part of strict), `e` is unknown
function toMessage(e: unknown): string {
  if (e instanceof Error) return e.message;
  if (typeof e === "string") return e;
  return JSON.stringify(e);
}

class HttpError extends Error {
  constructor(public readonly status: number, message: string) {
    super(message);
    this.name = "HttpError";
  }
}

for (const thrower of [() => { throw new HttpError(404, "Not found"); }, () => { throw "text"; }, () => { throw { code: 1 }; }]) {
  try { thrower(); }
  catch (e) {
    if (e instanceof HttpError) console.log("HTTP", e.status, e.message);
    else console.log("other:", toMessage(e));
  }
}

// Result pattern: make failures part of the type
type Ok<T> = { ok: true; value: T };
type Err<E> = { ok: false; error: E };
type Result<T, E> = Ok<T> | Err<E>;

function divide(a: number, b: number): Result<number, "div-by-zero"> {
  return b === 0 ? { ok: false, error: "div-by-zero" } : { ok: true, value: a / b };
}
const r = divide(4, 0);
if (r.ok) console.log(r.value); else console.log("error:", r.error);

19. Common Pitfalls

// 1. Types are erased: no runtime checks
function parseAge(raw: unknown): number {
  return raw as number;                        // lies if raw is a string
}
console.log(typeof parseAge("12"));            // "string" at runtime!

// 2. Object.keys returns string[]
const o = { a: 1, b: 2 };
const keys = Object.keys(o) as (keyof typeof o)[];
console.log(keys.map((k) => o[k]));

// 3. Structural typing allows extra props via variables
interface Pt { x: number }

20. Practice Exercises

1. Typed event emitter

type Handler<T> = (payload: T) => void;

class Emitter<Events extends Record<string, unknown>> {
  private handlers: { [K in keyof Events]?: Handler<Events[K]>[] } = {};

  on<K extends keyof Events>(event: K, fn: Handler<Events[K]>): void {
    (this.handlers[event] ??= []).push(fn);
  }
  emit<K extends keyof Events>(event: K, payload: Events[K]): void {
    this.handlers[event]?.forEach((fn) => fn(payload));
  }
}

const bus = new Emitter<{ login: { user: string }; logout: void }>();
bus.on("login", ({ user }) => console.log("welcome", user));
bus.emit("login", { user: "Ann" });

2. Typed groupBy

function groupBy<T, K extends PropertyKey>(items: readonly T[], keyFn: (item: T) => K): Record<K, T[]> {
  const out = {} as Record<K, T[]>;
  for (const item of items) {
    (out[keyFn(item)] ??= []).push(item);
  }
  return out;
}
const people = [{ n: "A", age: 20 }, { n: "B", age: 30 }, { n: "C", age: 20 }];
console.log(groupBy(people, (p) => p.age));

3. Deep partial

type DeepPartial<T> = T extends object ? { [K in keyof T]?: DeepPartial<T[K]> } : T;

interface Settings { ui: { theme: string; font: { size: number; family: string } }; lang: string }

function merge<T>(base: T, patch: DeepPartial<T>): T {
  const out: any = { ...base };
  for (const [k, v] of Object.entries(patch as object)) {
    out[k] = v && typeof v === "object" ? merge((base as any)[k], v) : v;
  }
  return out;
}
const base: Settings = { ui: { theme: "dark", font: { size: 12, family: "Arial" } }, lang: "en" };
console.log(merge(base, { ui: { font: { size: 14 } } }));

4. State machine with discriminated unions

type State =
  | { status: "idle" }
  | { status: "loading" }
  | { status: "success"; data: string[] }
  | { status: "error"; message: string };

type Action =
  | { type: "fetch" }
  | { type: "resolve"; data: string[] }
  | { type: "reject"; message: string };

function reducer(state: State, action: Action): State {
  switch (action.type) {
    case "fetch": return { status: "loading" };
    case "resolve": return { status: "success", data: action.data };
    case "reject": return { status: "error", message: action.message };
  }
}

let s: State = { status: "idle" };
for (const a of [{ type: "fetch" }, { type: "resolve", data: ["x"] }] as Action[]) {
  s = reducer(s, a);
  console.log(s);
}
